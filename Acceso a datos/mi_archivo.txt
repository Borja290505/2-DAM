3. Buscar los libros de un autor específico ej: autor con id 1
db.libros.aggregate([
	{
		$lookup:{
			from: "autores",
			localField: "autor_id",
			foreignField: "_id",
			as: "info_autor"
		}
	},
	{
		$match: {autor_id:1}
	}
])

7. Busca la información relativa a los préstamos existentes, pero debes mostrar la
información de los libros, autores y usuarios.
db.libros.aggregate([
	{
		$lookup:{
			from: "autores",
			localField: "autor_id",
			foreignField: "_id",
			as: "autores"
		}
	},
	{
		$lookup:{
			from: "editoriales",
			localField: "editorial_id",
			foreignField: "_id",
			as: "editorial"
		}
	},
	{
		$unwind: "$autores"
	},
	{
		$lookup:{
			from: "editoriales",
			localField: "usuario_id",
			foreignField: "_id",
			as: "usuario"
		}
	},
	{
		$unwind: "$usuario"
	},
	{
		$lookup:{
			from: "libros",
			localField: "libro_id",
			foreignField: "_id",
			as: "libro"
		}
	},
	{
		$unwind: "$libro"
	}
])

12. Encuentra los usuarios Premium con sus préstamos activos.
db.usuarios.aggregate([
	{
		$match: {tipo:"premium"}
	},
	{
		$lookup:{
			from:"prestamos",
			localField:"_id",
			foreignField:"usuario_id",
			as: "prestamos"
		}
	},
	{
		$unwind: "$prestamos"
	},
	{
		$lookup:{
			from: "libros",
			localField: "prestamos.libro_id",
			foreignField: "libro_id",
			as: "libro"
		}
	},
	{
		$unwind: "$libro"
	}
])
